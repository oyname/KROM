cmake_minimum_required(VERSION 3.22)
project(KromProject LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
find_package(Krom CONFIG REQUIRED)
if(NOT KROM_PROJECT_ROOT OR NOT EXISTS "${KROM_PROJECT_ROOT}/krom-project.json")
    message(FATAL_ERROR "Set KROM_PROJECT_ROOT to the project containing krom-project.json")
endif()
add_compile_options(/utf-8 /permissive- /Zc:__cplusplus)
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
krom_add_project_script_module("${KROM_PROJECT_ROOT}")
krom_read_project_runtime_target("${KROM_PROJECT_ROOT}" _project)
foreach(_backend IN ITEMS dx11 vulkan opengl)
    if(TARGET Krom::engine_${_backend})
        string(TOUPPER "${_backend}" _upper)
        krom_add_project_runtime("krom_runtime_${_project}_${_backend}" "${_project}_${_backend}"
            Krom::engine_platform_win32 Krom::engine_${_backend} "KROM_APP_BACKEND_${_upper}" "${KROM_PROJECT_ROOT}")
    endif()
endforeach()
